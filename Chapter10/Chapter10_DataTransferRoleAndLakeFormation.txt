# Use AWS Cloud Shell
# Step 1: Create the trust relationship policy file
echo '{
    "Version": "2012-10-17",
    "Statement": [{
        "Effect": "Allow",
        "Principal": {
        "Service": [
        "redshift.amazonaws.com",
        "glue.amazonaws.com"
        ]
    },
        "Action": "sts:AssumeRole"
    }]
}' > trust-policy.json

# Step 2: Create the IAM role
aws iam create-role --role-name DataTransferRole --assume-role-policy-document file://trust-policy.json

# Step 3: Create the policy file
echo '{
    "Version": "2012-10-17",
    "Statement": [
        {
        "Sid": "DataTransferRolePolicy",
        "Effect": "Allow",
        "Action": [
            "s3:GetObject",
            "s3:PutObject",
            "s3:DeleteObject",
            "s3:ListBucket",
            "redshift-data:GetStagingBucketLocation",
            "redshift-serverless:GetNamespace",
            "redshift-serverless:GetWorkgroup",
            "kms:GenerateDataKey",
            "kms:Decrypt",
            "glue:GetDatabase",
            "glue:GetCatalog"
        ],
        "Resource": [
            "*"
        ]
      }
    ]
}' > data-transfer-policy.json

# Step 4: Attach the inline policy to the role
aws iam put-role-policy --role-name DataTransferRole --policy-name DataTransferPolicy --policy-document file://data-transfer-policy.json
